package com.example.stickerbridge.processor

import android.content.Context
import android.graphics.Bitmap
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Matrix
import android.graphics.Paint
import android.net.Uri
import android.os.Build
import com.arthenica.ffmpegkit.FFmpegKit
import com.arthenica.ffmpegkit.ReturnCode
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.io.FileOutputStream
import kotlin.math.min

class StickerProcessor(private val context: Context) {

    private val subjectRemover = SubjectRemover()

    /**
     * Process static image: strips background (if removeBackground=true),
     * scales to full 512x512 canvas, and compresses to lossless WebP.
     */
    suspend fun processStaticSticker(
        sourceBitmap: Bitmap,
        fileName: String,
        removeBackground: Boolean = false
    ): File? = withContext(Dispatchers.IO) {
        val targetBitmap = if (removeBackground) {
            subjectRemover.extractSubject(sourceBitmap) ?: sourceBitmap
        } else {
            sourceBitmap
        }

        val standardized = fitToWhatsAppCanvas(targetBitmap)
        val stickersDir = File(context.filesDir, "stickers").apply { if (!exists()) mkdirs() }
        val outputFile = File(stickersDir, "$fileName.webp")

        FileOutputStream(outputFile).use { outStream ->
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                standardized.compress(Bitmap.CompressFormat.WEBP_LOSSLESS, 100, outStream)
            } else {
                @Suppress("DEPRECATION")
                standardized.compress(Bitmap.CompressFormat.WEBP, 90, outStream)
            }
        }
        outputFile
    }

    /**
     * Converts a Video / GIF directly to 512x512 Animated WebP under 500 KB.
     */
    suspend fun processAnimatedSticker(sourceUri: Uri, fileName: String): File? = withContext(Dispatchers.IO) {
        val stickersDir = File(context.filesDir, "stickers").apply { if (!exists()) mkdirs() }
        val outputFile = File(stickersDir, "$fileName.webp")

        // Copy input URI to a temporary cache file so FFmpeg can read it directly
        val tempInput = File(context.cacheDir, "temp_in_${System.currentTimeMillis()}")
        context.contentResolver.openInputStream(sourceUri)?.use { input ->
            tempInput.outputStream().use { output -> input.copyTo(output) }
        }

        // WhatsApp animated specs: max 512x512, <= 30 fps, looping (loop 0), scaled preserving aspect ratio
        val cmd = "-y -i \"${tempInput.absolutePath}\" " +
                "-vf \"scale=512:512:force_original_aspect_ratio=decrease,pad=512:512:(512-iw)/2:(512-ih)/2:color=0x00000000,fps=15\" " +
                "-vcodec libwebp -lossless 0 -compression_level 4 -q:v 60 -loop 0 -an \"${outputFile.absolutePath}\""

        val session = FFmpegKit.execute(cmd)
        tempInput.delete()

        if (ReturnCode.isSuccess(session.returnCode)) {
            outputFile
        } else {
            null
        }
    }

    /**
     * Scale to fill the 512x512 canvas edge-to-edge without squishing or down-padding.
     */
    private fun fitToWhatsAppCanvas(src: Bitmap): Bitmap {
        val targetSize = 512
        val output = Bitmap.createBitmap(targetSize, targetSize, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(output)

        // Exact proportional fit up to 512
        val scale = min(targetSize.toFloat() / src.width, targetSize.toFloat() / src.height)
        val scaledWidth = src.width * scale
        val scaledHeight = src.height * scale

        val left = (targetSize - scaledWidth) / 2f
        val top = (targetSize - scaledHeight) / 2f

        val paint = Paint(Paint.ANTI_ALIAS_FLAG or Paint.FILTER_BITMAP_FLAG)
        val matrix = Matrix().apply {
            postScale(scale, scale)
            postTranslate(left, top)
        }

        canvas.drawBitmap(src, matrix, paint)
        return output
    }

    fun ensureTrayIconExists() {
        val stickersDir = File(context.filesDir, "stickers").apply { if (!exists()) mkdirs() }
        val trayFile = File(stickersDir, "tray_icon.png")
        if (!trayFile.exists()) {
            val iconBitmap = Bitmap.createBitmap(96, 96, Bitmap.Config.ARGB_8888)
            val canvas = Canvas(iconBitmap)
            val paint = Paint().apply { color = Color.BLUE }
            canvas.drawCircle(48f, 48f, 44f, paint)
            FileOutputStream(trayFile).use { out ->
                iconBitmap.compress(Bitmap.CompressFormat.PNG, 100, out)
            }
        }
    }
}

package com.example.stickerbridge.processor


import android.graphics.Bitmap
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.segmentation.subject.SubjectSegmentation
import com.google.mlkit.vision.segmentation.subject.SubjectSegmenterOptions
import kotlinx.coroutines.tasks.await

class SubjectRemover {

    private val options = SubjectSegmenterOptions.Builder()
        .enableForegroundBitmap()
        .build()

    private val segmenter = SubjectSegmentation.getClient(options)

    suspend fun extractSubject(bitmap: Bitmap): Bitmap? {
        val inputImage = InputImage.fromBitmap(bitmap, 0)
        val result = segmenter.process(inputImage).await()
        return result.foregroundBitmap
    }
}

data class Sticker(
    val imageFileName: String,
    val emojis: List<String> = emptyList()

        parcel.readString() ?: "",
        parcel.createStringArrayList() ?: emptyList()
    )

    override fun writeToParcel(parcel: Parcel, flags: Int) {
        parcel.writeString(imageFileName)
        parcel.writeStringList(emojis)
    }

    override fun describeContents(): Int = 0

    companion object CREATOR : Parcelable.Creator<Sticker> {
        override fun createFromParcel(parcel: Parcel): Sticker = Sticker(parcel)
        override fun newArray(size: Int): Array<Sticker?> = arrayOfNulls(size)
    }

import android.content.ContentProvider
import android.content.ContentValues
import android.content.UriMatcher
import android.content.res.AssetFileDescriptor
import android.database.Cursor
import android.database.MatrixCursor
import android.net.Uri
import android.os.ParcelFileDescriptor
import java.io.File

class StickerContentProvider : ContentProvider() {

    companion object {
        const val AUTHORITY = "com.tsosang.stickerbridge.stickercontentprovider"
        private const val METADATA_CODE = 1
        private const val METADATA_PACK_CODE = 2
        private const val STICKERS_CODE = 3
        private const val STICKERS_ASSET_CODE = 4

        private val MATCHER = UriMatcher(UriMatcher.NO_MATCH).apply {
            addURI(AUTHORITY, "metadata", METADATA_CODE)
            addURI(AUTHORITY, "metadata/*", METADATA_PACK_CODE)
            addURI(AUTHORITY, "stickers/*", STICKERS_CODE)
            addURI(AUTHORITY, "stickers_asset/*/*", STICKERS_ASSET_CODE)
        }
    }

    override fun onCreate(): Boolean = true

    override fun query(
        uri: Uri,
        projection: Array<out String>?,
        selection: String?,
        selectionArgs: Array<out String>?,
        sortOrder: String?
    ): Cursor? {
        val match = MATCHER.match(uri)
        return when (match) {
            METADATA_CODE -> getPackMetadataCursor()
            STICKERS_CODE -> getStickersCursor(uri.lastPathSegment ?: "")
            else -> null
        }
    }

    override fun openAssetFile(uri: Uri, mode: String): AssetFileDescriptor? {
        if (MATCHER.match(uri) == STICKERS_ASSET_CODE) {
            val segments = uri.pathSegments
            val fileName = segments[2]
            val file = File(context?.filesDir, "stickers/$fileName")
            if (file.exists()) {
                val pfd = ParcelFileDescriptor.open(file, ParcelFileDescriptor.MODE_READ_ONLY)
                return AssetFileDescriptor(pfd, 0, AssetFileDescriptor.UNKNOWN_LENGTH)
            }
        }
        return super.openAssetFile(uri, mode)
    }

    override fun getType(uri: Uri): String? = when (MATCHER.match(uri)) {
        METADATA_CODE -> "vnd.android.cursor.dir/vnd.$AUTHORITY.metadata"
        METADATA_PACK_CODE -> "vnd.android.cursor.item/vnd.$AUTHORITY.metadata"
        STICKERS_CODE -> "vnd.android.cursor.dir/vnd.$AUTHORITY.stickers"
        STICKERS_ASSET_CODE -> "image/webp"
        else -> null
    }

    override fun insert(uri: Uri, values: ContentValues?): Uri? = null
    override fun delete(uri: Uri, selection: String?, selectionArgs: Array<out String>?): Int = 0
    override fun update(uri: Uri, values: ContentValues?, s: String?, sArgs: Array<out String>?): Int = 0

    private fun getPackMetadataCursor(): Cursor {
        val cursor = MatrixCursor(arrayOf(
            "sticker_pack_identifier",
            "sticker_pack_name",
            "sticker_pack_publisher",
            "sticker_pack_icon",
            "android_play_store_link",
            "ios_app_store_link",
            "is_animated_pack"
        ))
        // 1 marks it as animated pack, 0 as static
        cursor.addRow(arrayOf<Any>("tiktok_imports", "TikTok Imports", "StickerBridge", "tray_icon.png", "", "", 1))
        return cursor
    }

    private fun getStickersCursor(packId: String): Cursor {
        val cursor = MatrixCursor(arrayOf("sticker_file_name", "sticker_emoji"))
        val dir = File(context?.filesDir, "stickers")
        dir.listFiles { f -> f.extension == "webp" }?.forEach { file ->
            cursor.addRow(arrayOf(file.name, "âš¡"))
        }
        return cursor
    }
}

        activity: Activity,

        packName: String = "TikTok Imports"
    ) {

            activity.startActivityForResult(intent, ADD_PACK_REQUEST_CODE)
        } catch (_: ActivityNotFoundException) {
            Toast.makeText(activity, "WhatsApp is not installed", Toast.LENGTH_SHORT).show()
        }
    }

@Composable
fun StickerBridgeScreen(processor: StickerProcessor) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    var isProcessing by remember { mutableStateOf(false) }
    var autoRemoveBg by remember { mutableStateOf(false) }

    val stickersDir = remember { File(context.filesDir, "stickers") }
    var stickerFiles by remember {
        mutableStateOf(stickersDir.listFiles { f -> f.extension == "webp" }?.toList() ?: emptyList())
    }

    // Modern ActivityResultLauncher for adding stickers to WhatsApp
    val whatsappLauncher = rememberLauncherForActivityResult(
        contract = ActivityResultContracts.StartActivityForResult()
    ) { result ->
        if (result.resultCode == Activity.RESULT_OK) {
            Toast.makeText(context, "Added to WhatsApp!", Toast.LENGTH_SHORT).show()
        }
    }

    // Media Picker supporting Images, GIFs, and Videos
    val mediaPickerLauncher = rememberLauncherForActivityResult(
        contract = ActivityResultContracts.GetContent()
    ) { uri: Uri? ->
        uri?.let { selectedUri ->
            scope.launch {
                isProcessing = true
                val mimeType = context.contentResolver.getType(selectedUri) ?: ""
                val timestamp = System.currentTimeMillis()

                val resultFile = if (mimeType.startsWith("video/") || mimeType.contains("gif")) {
                    processor.processAnimatedSticker(selectedUri, "sticker_$timestamp")
                } else {
                    context.contentResolver.openInputStream(selectedUri)?.use { stream ->
                        val bitmap = BitmapFactory.decodeStream(stream)
                        processor.processStaticSticker(bitmap, "sticker_$timestamp", removeBackground = autoRemoveBg)
                    }
                }

                if (resultFile != null && resultFile.exists()) {
                    stickerFiles = stickersDir.listFiles { f -> f.extension == "webp" }?.toList() ?: emptyList()
                    Toast.makeText(context, "Sticker imported!", Toast.LENGTH_SHORT).show()
                } else {
                    Toast.makeText(context, "Failed to convert sticker", Toast.LENGTH_SHORT).show()
                }
                isProcessing = false
            }
        }
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Text("StickerBridge", style = MaterialTheme.typography.headlineMedium)
        Text(
            text = "${stickerFiles.size} stickers ready (WhatsApp requires at least 3)",
            style = MaterialTheme.typography.bodyMedium,
            color = Color.Gray,
            modifier = Modifier.padding(bottom = 8.dp)
        )

        Row(
            verticalAlignment = Alignment.CenterVertically,
            modifier = Modifier.padding(bottom = 12.dp)
        ) {
            Checkbox(checked = autoRemoveBg, onCheckedChange = { autoRemoveBg = it })
            Text("Auto-remove background (ML Kit)")
        }

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            Button(
                onClick = { mediaPickerLauncher.launch("*/*") },
                modifier = Modifier.weight(1f),
                enabled = !isProcessing
            ) {
                Text(if (isProcessing) "Converting..." else "Import Media")
            }

            Button(
                onClick = {
                    if (stickerFiles.size < 3) {
                        Toast.makeText(context, "Add at least 3 stickers first", Toast.LENGTH_SHORT).show()
                    } else {
                        val intent = Intent("com.whatsapp.intent.action.ENABLE_STICKER_PACK").apply {
                            putExtra("sticker_pack_id", "tiktok_imports")
                            putExtra("sticker_pack_authority", StickerContentProvider.AUTHORITY)
                            putExtra("sticker_pack_name", "TikTok Imports")
                        }
                        try {
                            whatsappLauncher.launch(intent)
                        } catch (_: Exception) {
                            Toast.makeText(context, "WhatsApp not found or permission error", Toast.LENGTH_LONG).show()
                        }
                    }
                },
                modifier = Modifier.weight(1f),
                enabled = !isProcessing && stickerFiles.size >= 3
            ) {
                Text("Add to WhatsApp")
            }
        }

        Spacer(modifier = Modifier.height(16.dp))

        if (isProcessing) {
            CircularProgressIndicator(modifier = Modifier.padding(16.dp))
        }

        LazyVerticalGrid(
            columns = GridCells.Fixed(3),
            verticalArrangement = Arrangement.spacedBy(8.dp),
            horizontalArrangement = Arrangement.spacedBy(8.dp),
            modifier = Modifier.fillMaxSize()
        ) {
            items(stickerFiles) { file ->
                Box(
                    modifier = Modifier
                        .aspectRatio(1f)
                        .clip(RoundedCornerShape(8.dp))
                        .background(Color(0xFFEEEEEE)),
                    contentAlignment = Alignment.Center
                ) {
                    AsyncImage(
                        model = file,
                        contentDescription = "Sticker preview",
                        modifier = Modifier.fillMaxSize()
                    )
                }
            }
        }
    }
}
